title = "Стек"
dialogue = [
    (mentor_left, `Задача этого уровня — реализовать распространённую структуру данных под названием «стек». Вы можете делать с ним две вещи:

[color=#66ccff]PUSH[/color] — добавляет число в стек
[color=#66ccff]POP[/color]  — убирает последнее добавленное значение из стека

Стек можно представить как стопку тарелок в ресторане в час пик. Мойщик посуды [color=#66ccff]кладёт[/color] тарелки сверху, а повар [color=#66ccff]берёт[/color] их сверху.

Прочитайте вход на этом уровне, чтобы узнать следующую операцию.
Ненулевой вход — это PUSH, в этом случае сохраните значение, чтобы потом его можно было извлечь.
Значение 0 — это POP, уберите и выведите последнее сохранённое число.
`),
]
post_mortem = [
  (mentor_left, `Поскольку стек очень полезен для программирования, в следующий уровень Symphony будут добавлены инструкции «push» и «pop». Обе они — синонимы других инструкций и используют регистр «sp» (сокращение от stack pointer), чтобы отслеживать стек.

«push» — синоним:
sub sp, sp, 4
store_32 [sp], %value

«pop» — синоним:
load_32 %destination, [sp]
add sp, sp, 4

Обратите внимание, что «push» вычитает из «sp», поэтому стек растёт вниз от старших адресов. Младшие адреса остаются свободными для ваших нужд.`),
]
